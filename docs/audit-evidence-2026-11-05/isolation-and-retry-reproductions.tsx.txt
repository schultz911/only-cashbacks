function snapshot(data: any) {
  return { metadata: { hasPendingWrites: false }, exists: () => true, data: () => structuredClone(data) };
}

async function advance(ms: number) {
  for (let elapsed = 0; elapsed < ms; elapsed += 100) {
    await act(async () => { await vi.advanceTimersByTimeAsync(Math.min(100, ms - elapsed)); });
  }
}
async function hydrate(data = cloudWallet()) {
  await act(async () => { state.snapshot(snapshot(data)); });
}
function deferred<T>() {
  let resolve!: (value: T) => void;
  const promise = new Promise<T>(done => { resolve = done; });
  return { promise, resolve };
}

describe('readiness sweep isolated reproductions', () => {
  it('checks whether an absent second-account document keeps first-account secrets and logs', async () => {
    mount();
    await hydrate({ ...cloudWallet(), openRouterApiKey: 'sk-or-synthetic-account-a', cashbackLogs: [{date: 123, amount: 42, cardId: 'sbi-cashback', cardName: 'A', merchantName: 'private-a', category: 'Other'}] });
    await advance(1000);
    await act(async () => { state.authChanged({uid: 'synthetic-account-b'}); });
    await act(async () => { state.snapshot({metadata:{hasPendingWrites:false},exists:()=>false}); });
    expect(search.openRouterApiKey).toBe('sk-or-synthetic-account-a');
    expect(wallet.cashbackLogs[0].merchantName).toBe('private-a');
    await act(async () => { sync.setTheme('dark'); });
    await advance(2200);
    expect(state.writes.at(-1)).toMatchObject({uid:'synthetic-account-b',data:{openRouterApiKey:'sk-or-synthetic-account-a',cashbackLogs:[{merchantName:'private-a'}]}});
    console.log('AUDIT_CROSS_ACCOUNT', JSON.stringify({secondAccount:state.writes.at(-1)?.uid, inheritedKey:state.writes.at(-1)?.data.openRouterApiKey, inheritedLogs:state.writes.at(-1)?.data.cashbackLogs.length}));
  });
  it('checks whether a global offline queue merges first-account logs into a second existing account', async () => {
    mount(); await hydrate(); await advance(1000);
    state.queue = [{date:456,amount:99,cardId:'sbi-cashback',cardName:'A',merchantName:'private-account-a',category:'Other'}];
    await act(async () => {state.authChanged({uid:'synthetic-account-b'});});
    await hydrate({...cloudWallet(),walletCards:['sbi-eternal']});
    await advance(2200);
    expect(state.writes.at(-1)).toMatchObject({uid:'synthetic-account-b',data:{cashbackLogs:[{merchantName:'private-account-a'}]}});
    console.log('AUDIT_QUEUE_ACCOUNT', JSON.stringify({secondAccount:state.writes.at(-1)?.uid,inheritedLogs:state.writes.at(-1)?.data.cashbackLogs.length}));
  });
  it('counts autosave retries after a persistent permission error with no additional edits', async () => {
    mount(); await hydrate(); await advance(1000);
    const calls = vi.mocked(setDoc).mock.calls.length;
    vi.mocked(setDoc).mockImplementation(() => new Promise((_resolve,reject) => setTimeout(() => reject(new Error('permission-denied')),100)));
    await act(async () => {sync.setTheme('dark');});
    await advance(12000);
    const attempted=vi.mocked(setDoc).mock.calls.length-calls;
    expect(attempted).toBeGreaterThanOrEqual(10);
    console.log('AUDIT_RETRY_STORM',JSON.stringify({attempts:attempted,simulatedSeconds:12,additionalEdits:0}));
  });
});
